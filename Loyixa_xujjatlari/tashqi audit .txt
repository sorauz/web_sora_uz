1.1. Tovar sahifasi uchun to‘g‘ri generateMetadata (app/[locale]/products/[slug]/page.tsx)
Root layout.tsx faylida title.template: "%s | Sora.uz" belgilangan bo‘lsa, ichki sahifalarda | Sora.uz yozish shart emas. Shuningdek, har bir sahifada canonical va ikki tilli alternates.languages (uz-UZ, ru-UZ) aniq ko‘rsatilishi zarur:

import type { Metadata } from 'next';

interface Props {
  params: { locale: 'uz' | 'ru'; slug: string };
}

export async function generateMetadata({ params }: Props): Promise<Metadata> {
  const { locale, slug } = params;
  const product = await getProductBySlug(slug, locale); // 1C / DB dan ma'lumot olish

  if (!product) return {};

  const isRu = locale === 'ru';
  const baseUrl = 'https://sora.uz';

  // Tilga mos SEO Title va Description shablonlari (ortiqcha "| Sora.uz" siz)
  const title = isRu
    ? `${product.nameRu} (SKU: ${product.sku}) — купить по оптовой цене в Ташкенте`
    : `${product.nameUz} (SKU: ${product.sku}) — Toshkentda ulgurji narxda xarid qilish`;

  const description = isRu
    ? `${product.shortDescriptionRu || product.nameRu}. Официальная гарантия 1С ERP, работаем по договору и ЭСФ Didox. Доставка по Ташкенту за 24 часа.`
    : `${product.shortDescriptionUz || product.nameUz}. Rasmiy 1C ERP kafolati, shartnoma va Didox EHF orqali xarid. Toshkent bo'ylab 24 soatda yetkazib berish.`;

  const currentPath = isRu ? `/ru/products/${slug}` : `/products/${slug}`;

  return {
    title, // Layout template avtomatik "| Sora.uz" ni qo'shadi
    description,
    alternates: {
      canonical: `${baseUrl}${currentPath}`,
      languages: {
        'uz-UZ': `${baseUrl}/products/${product.slugUz || slug}`,
        'ru-UZ': `${baseUrl}/ru/products/${product.slugRu || slug}`,
        'x-default': `${baseUrl}/products/${product.slugUz || slug}`,
      },

    openGraph: {
      title,
      description,
      url: `${baseUrl}${currentPath}`,

      images: [
        {
          url: product.imageUrl || `${baseUrl}/og-default.jpg`,
          width: 800,
          height: 800,
          alt: isRu ? product.nameRu : product.nameUz,

1.2. Kategoriya va Brend paginatsiyalarini tartibga solish (Crawl Budget himoyasi)
Google qidiruvida /ru/brand/deli?page=34 kabi sahifalar asosiy sahifa bilan raqobatlashmasligi uchun 1-sahifadan keyingi barcha paginatsiya va filtr parametrlariga noindex, follow qoidasini berish tavsiya etiladi:

``
// app/[locale]/brand/[slug]/page.tsx yoki category/[slug]/page.tsx ichida:
export async function generateMetadata({
  params,
  searchParams,
}: {
  params: { locale: 'uz' | 'ru'; slug: string };
  searchParams: { page?: string; sort?: string; price?: string };
}): Promise<Metadata> {
  const pageNum = Number(searchParams.page) || 1;
  const hasFilters = Boolean(searchParams.sort || searchParams.price);
  const isRu = params.locale === 'ru';
  const cleanUrl = `https://sora.uz${isRu ? '/ru' : ''}/brand/${params.slug}`;

  return {
    // Canonical har doim parametrsiz 1-sahifaga yoki o'zining toza URL'iga qaraydi
    alternates: {
      canonical: cleanUrl,
    },
    // Agar page 1 dan katta bo'lsa yoki filtr ishlasa: bot linklarni o'qiydi (follow), lekin sahifani indekslamaydi (noindex)
    robots:
      pageNum > 1 || hasFilters
        ? { index: false, follow: true }
        : { index: true, follow: true },
  };
}
``

3. Tovar sahifasi uchun Mukammal @graph JSON-LD Komponenti
Tovar sahifasida alohida-alohida skriptlar o‘rniga Product, BreadcrumbList va FAQPage sxemalarini yagona @graph ichida birlashtirib berish eng yuqori SEO samaradorlikni beradi.

Quyidagi tayyor komponentni tovar sahifasiga (app/[locale]/products/[slug]/page.tsx) joylashtirishingiz mumkin:

``
// components/seo/ProductJsonLd.tsx
import React from 'react';

interface ProductData {
  id: string;
  sku: string;
  gtin13?: string; // 1C dagi EAN-13 shtrix-kod (masalan: "6942625454087")
  name: string;
  description: string;
  slug: string;
  brandName: string;
  categoryName: string;
  categorySlug: string;
  price: number; // 0 bo'lsa - "Narxi kelishiladi"
  stockCount: number; // 1C ombor qoldig'i
  images: string[];
  ratingValue?: number;
  reviewCount?: number;
}

export function ProductJsonLd({
  product,
  locale,
}: {
  product: ProductData;
  locale: 'uz' | 'ru';
}) {
  const isRu = locale === 'ru';
  const baseUrl = 'https://sora.uz';
  const productUrl = `${baseUrl}${isRu ? '/ru' : ''}/products/${product.slug}`;
  const categoryUrl = `${baseUrl}${isRu ? '/ru' : ''}/category/${product.categorySlug}`;

  // EAN-13 shtrix-kod tekshiruvi (faqat 13 ta raqamdan iborat bo'lsa qo'shiladi)
  const isValidGtin13 = Boolean(
    product.gtin13 && /^\d{13}$/.test(product.gtin13.trim())
  );

  // Narx mavjud bo'lgandagina to'liq Offer obyektini shakllantiramiz
  const offerObject =
    product.price > 0
      ? {
          '@type': 'Offer',
          url: productUrl,
          priceCurrency: 'UZS',
          price: product.price.toString(),
          priceValidUntil: '2027-12-31',
          itemCondition: 'https://schema.org/NewCondition',
          availability:
            product.stockCount > 0
              ? 'https://schema.org/InStock'
              : 'https://schema.org/OutOfStock',
          inventoryLevel: {
            '@type': 'QuantitativeValue',
            value: product.stockCount,
          },
          seller: {
            '@type': 'Organization',
            name: 'SORAUZ MCHJ',
            url: baseUrl,
          },
          // Yetkazib berish shartlari (500 000 so'mdan yuqori bepul, aks holda 30 000 so'm)
          shippingDetails: {
            '@type': 'OfferShippingDetails',
            shippingRate: {
              '@type': 'MonetaryAmount',
              value: product.price >= 500000 ? '0' : '30000',
              currency: 'UZS',
            },
            shippingDestination: {
              '@type': 'DefinedRegion',
              addressCountry: 'UZ',
            },
            deliveryTime: {
              '@type': 'ShippingDeliveryTime',
              handlingTime: {
                '@type': 'QuantitativeValue',
                minValue: 0,
                maxValue: 1,
                unitCode: 'DAY',
              },
              transitTime: {
                '@type': 'QuantitativeValue',
                minValue: 1,
                maxValue: 3,
                unitCode: 'DAY',
              },
            },
          },
          // Qaytarish siyosati (14 kun)
          hasMerchantReturnPolicy: {
            '@type': 'MerchantReturnPolicy',
            applicableCountry: 'UZ',
            returnPolicyCategory:
              'https://schema.org/MerchantReturnFiniteReturnWindow',
            merchantReturnDays: 14,
            returnMethod: 'https://schema.org/ReturnInStore',
            returnFees: 'https://schema.org/FreeReturn',
          },
        }
      : undefined;

  const jsonLd = {
    '@context': 'https://schema.org',
    '@graph': [
      // 1. BreadcrumbList (Navigatsiya zanjiri)
      {
        '@type': 'BreadcrumbList',
        '@id': `${productUrl}#breadcrumb`,
        itemListElement: [
          {
            '@type': 'ListItem',
            position: 1,
            name: isRu ? 'Главная' : 'Bosh sahifa',
            item: `${baseUrl}${isRu ? '/ru' : ''}`,
          },
          {
            '@type': 'ListItem',
            position: 2,
            name: isRu ? 'Каталог' : 'Katalog',
            item: `${baseUrl}${isRu ? '/ru/catalog' : '/catalog'}`,
          },
          {
            '@type': 'ListItem',
            position: 3,
            name: product.categoryName,
            item: categoryUrl,
          },
          {
            '@type': 'ListItem',
            position: 4,
            name: product.name,
            item: productUrl,
          },
        ],
      },

      // 2. Product (Asosiy mahsulot mikro-belgisi)
      {
        '@type': 'Product',
        '@id': `${productUrl}#product`,
        name: product.name,
        description: product.description,
        image: product.images,
        sku: product.sku,
        mpn: product.sku,
        ...(isValidGtin13 ? { gtin13: product.gtin13!.trim() } : {}),
        brand: {
          '@type': 'Brand',
          name: product.brandName,
        },
        ...(offerObject ? { offers: offerObject } : {}),
        // Faqat haqiqiy izohlar mavjud bo'lganda qo'shiladi
        ...(product.reviewCount && product.reviewCount > 0 && product.ratingValue
          ? {
              aggregateRating: {
                '@type': 'AggregateRating',
                ratingValue: product.ratingValue.toString(),
                reviewCount: product.reviewCount.toString(),
                bestRating: '5',
                worstRating: '1',
              },
            }
          : {}),
      },
    ],
  };

  return (
    <script
      type="application/ld+json"
      dangerouslySetInnerHTML={{ __html: JSON.stringify(jsonLd) }}
    />
  );
}
``
